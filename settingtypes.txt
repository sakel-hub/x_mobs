# ==============================================================================
# X Mobs - Global Configuration
# ==============================================================================

# Enable natural mapgen and trickle mob spawning across the world.
# If false, ambient wilderness spawning is disabled; boss encounters, combat minion summons, and admin commands still work.
x_mobs.enable_spawning (Enable Natural Spawning) bool true

# Require players to carry the Bestiary of the Realm tome in their inventory to record sightings and kills.
x_mobs.bestiary_require_book_in_inventory (Require Bestiary Tome in Inventory) bool false

# When true, mobs will not naturally target players in creative mode unless attacked first.
x_mob_core.creative_peaceful (Creative Mode Neutrality) bool true
